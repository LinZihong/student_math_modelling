import pandas as pd
import numpy as np
from collections import defaultdict, deque
import heapq

# 1) Load data
excel_file = '/Users/(*file path*)/§{\cjkfont 附件}§1.xlsx'
coords_c = pd.read_excel(excel_file, sheet_name=0, header=0)
edges_c = pd.read_excel(excel_file, sheet_name=1, header=0)

node_ids = coords_c.iloc[:, 0].astype(str).values
X = coords_c.iloc[:, 1].astype(float).values
Y = coords_c.iloc[:, 2].astype(float).values
Z = coords_c.iloc[:, 3].astype(float).values

edge_ids = edges_c.iloc[:, 0].astype(str).values
node1_ids = edges_c.iloc[:, 1].astype(str).values
node2_ids = edges_c.iloc[:, 2].astype(str).values

# 2) Physical parameters
W = 4         # tunnel width (m)
H = 3         # tunnel height (m)
h0 = 0.1      # initial film height (m)
Q_burst = 30 # burst inflow (m³/min)
Q_in = Q_burst / 60 # m³/s

# First burst point
burst_pt1 = np.array([5349.03, 4931.90, 10.00]) # A1
burst_time1 = 0.0 # seconds

# Second burst point (starts 4 minutes after first)
burst_pt2 = np.array([3760.40, 3808.33, 10.00]) # B1
burst_time2 = 4 * 60 # seconds

# 3) Build initial graph
node_map = {str(node_id): idx for idx, node_id in enumerate(node_ids)}
num_nodes = len(node_ids)

# Build adjacency list and edge info
graph = defaultdict(list)
edge_info = {} # edge_id -> (node1, node2, length)
edge_lengths = {}

for i in range(len(edge_ids)):
    node1 = str(node1_ids[i])
    node2 = str(node2_ids[i])
    idx1 = node_map[node1]
    idx2 = node_map[node2]

    length = np.linalg.norm([X[idx2]-X[idx1], Y[idx2]-Y[idx1], Z[idx2]-Z[idx1]])
    edge_info[edge_ids[i]] = (node1, node2, length)
    edge_lengths[edge_ids[i]] = length

    graph[node1].append((node2, edge_ids[i], length))
    graph[node2].append((node1, edge_ids[i], length))

# 4) Function to add burst point to graph
def add_burst_point(burst_pt, burst_node_id, graph, edge_info, edge_lengths, node_map, X, Y, Z, node_ids):
    # Find closest edge to burst point
    best_edge = None
    best_position = 0.5
    best_distance = float('inf')

    for edge_id, (node1, node2, length) in edge_info.items():
        idx1 = node_map[node1]
        idx2 = node_map[node2]

        p1 = np.array([X[idx1], Y[idx1], Z[idx1]])
        p2 = np.array([X[idx2], Y[idx2], Z[idx2]])

        # Project burst point onto the edge
        ab = p2 - p1
        if np.linalg.norm(ab) < 1e-10:
            continue

        t = np.dot(burst_pt - p1, ab) / np.dot(ab, ab)
        t = max(0, min(1, t))
        proj = p1 + t * ab
        distance = np.linalg.norm(burst_pt - proj)

        if distance < best_distance:
            best_distance = distance
            best_edge = edge_id
            best_position = t
    if best_edge is None or best_distance > 1.0:
        raise ValueError(f"Could not find suitable burst point {burst_node_id} on any edge")

    # Add burst point as a new node
    new_node_idx = len(node_ids)
    node_map[burst_node_id] = new_node_idx
    node_ids = np.append(node_ids, burst_node_id)
    X = np.append(X, burst_pt[0])
    Y = np.append(Y, burst_pt[1])
    Z = np.append(Z, burst_pt[2])

    # Split the burst edge into two parts
    node1, node2, total_length = edge_info[best_edge]
    part1_length = best_position * total_length
    part2_length = (1 - best_position) * total_length

    # Remove original edge and add two new edges
    del edge_info[best_edge]
    del edge_lengths[best_edge]
    graph[node1] = [n for n in graph[node1] if n[1] != best_edge]
    graph[node2] = [n for n in graph[node2] if n[1] != best_edge]

    edge1_id = f"{best_edge}_part1_{burst_node_id}"
    edge2_id = f"{best_edge}_part2_{burst_node_id}"

    edge_info[edge1_id] = (node1, burst_node_id, part1_length)
    edge_info[edge2_id] = (burst_node_id, node2, part2_length)
    edge_lengths[edge1_id] = part1_length
    edge_lengths[edge2_id] = part2_length

    graph[node1].append((burst_node_id, edge1_id, part1_length))
    graph[burst_node_id].append((node1, edge1_id, part1_length))
    graph[burst_node_id].append((node2, edge2_id, part2_length))
    graph[node2].append((burst_node_id, edge2_id, part2_length))

    return burst_node_id, edge1_id, edge2_id, best_position, node_ids, X, Y, Z

# 5) Add both burst points sequentially
burst_node_id1, edge1_id1, edge2_id1, pos1, node_ids, X, Y, Z = add_burst_point(
    burst_pt1, "BurstPoint1", graph, edge_info, edge_lengths, node_map, X, Y, Z, node_ids
)

# Update node_map after first burst point addition
for i, node_id in enumerate(node_ids):
    node_map[node_id] = i

burst_node_id2, edge1_id2, edge2_id2, pos2, node_ids, X, Y, Z = add_burst_point(
    burst_pt2, "BurstPoint2", graph, edge_info, edge_lengths, node_map, X, Y, Z, node_ids
)

# Final update of node_map
for i, node_id in enumerate(node_ids):
    node_map[node_id] = i

num_nodes = len(node_ids)

# 6) Modified water propagation simulation with two burst points
edge_reach_time = {edge_id: float('inf') for edge_id in edge_info}
edge_fill_time = {edge_id: float('inf') for edge_id in edge_info}
node_reach_time = {node_id: float('inf') for node_id in node_map}

# Track the state of each edge
edge_water_height = {edge_id: 0.0 for edge_id in edge_info}
edge_is_filled = {edge_id: False for edge_id in edge_info}
edge_has_reached_min = {edge_id: False for edge_id in edge_info}
edge_flow_sources = {edge_id: [] for edge_id in edge_info} # Track multiple flow sources

# Priority queue for events: (time, event_type, edge_id, node_id, flow_rate, source_id)
pq = []

# Initialize first burst point
node_reach_time[burst_node_id1] = burst_time1
for neighbor, edge_id, length in graph[burst_node_id1]:
    flow_rate = Q_in / len(graph[burst_node_id1])
    volume_needed = W * h0 * length
    reach_time = burst_time1 + volume_needed / flow_rate
    heapq.heappush(pq, (reach_time, 'reach', edge_id, neighbor, flow_rate, burst_node_id1))

# Schedule second burst point
heapq.heappush(pq, (burst_time2, 'burst', '', burst_node_id2, Q_in, ''))

# Process events
while pq:
    current_time, event_type, edge_id, node_id, flow_rate, source_id = heapq.heappop(pq)

    if event_type == 'burst':
        # Second burst point starts
        node_reach_time[node_id] = current_time
        for neighbor, edge_id, length in graph[node_id]:
            flow_rate = Q_in / len(graph[node_id])
            volume_needed = W * h0 * length
            reach_time = current_time + volume_needed / flow_rate
            heapq.heappush(pq, (reach_time, 'reach', edge_id, neighbor, flow_rate, node_id))
    elif event_type == 'reach':
        if edge_has_reached_min[edge_id]:
            # Edge already reached minimum height, add flow from new source
            edge_flow_sources[edge_id].append((source_id, flow_rate))

            # Recalculate fill time with additional flow
            if not edge_is_filled[edge_id]:
                total_flow = sum(f for _, f in edge_flow_sources[edge_id])
                remaining_volume = W * (H - edge_water_height[edge_id]) * edge_lengths[edge_id]
                new_fill_time = current_time + remaining_volume / total_flow
                heapq.heappush(pq, (new_fill_time, 'fill', edge_id, node_id, total_flow, 'combined'))
            continue

        # Water reaches 0.1m height in this edge
        edge_has_reached_min[edge_id] = True
        edge_reach_time[edge_id] = current_time
        edge_water_height[edge_id] = h0
        edge_flow_sources[edge_id].append((source_id, flow_rate))

        # Update node reach time
        if current_time < node_reach_time[node_id]:
            node_reach_time[node_id] = current_time

        # Find all outgoing edges from this node
        outgoing_edges = []
        for next_neighbor, next_edge_id, next_length in graph[node_id]:
            if next_edge_id != edge_id and not edge_is_filled[next_edge_id]:
                outgoing_edges.append((next_edge_id, next_neighbor, next_length))

        if outgoing_edges:
            # Distribute flow equally among available outgoing edges
            total_flow = sum(f for _, f in edge_flow_sources[edge_id])
            flow_per_edge = total_flow / len(outgoing_edges)

            for next_edge_id, next_node, next_length in outgoing_edges:
                if not edge_has_reached_min[next_edge_id]:
                    volume_needed = W * h0 * next_length
                    next_reach_time = current_time + volume_needed / flow_per_edge
                    heapq.heappush(pq, (next_reach_time, 'reach', next_edge_id, next_node, flow_per_edge, source_id))
                else:
                    remaining_volume = W * (H - edge_water_height[next_edge_id]) * next_length
                    new_fill_time = current_time + remaining_volume / flow_per_edge
                    heapq.heappush(pq, (new_fill_time, 'fill', next_edge_id, next_node, flow_per_edge, source_id))
        # Schedule fill event for current edge
        total_flow = sum(f for _, f in edge_flow_sources[edge_id])
        additional_volume = W * (H - h0) * edge_lengths[edge_id]
        fill_time = current_time + additional_volume / total_flow
        heapq.heappush(pq, (fill_time, 'fill', edge_id, node_id, total_flow, 'combined'))

    elif event_type == 'fill':
        if edge_is_filled[edge_id]:
            continue

        edge_is_filled[edge_id] = True
        edge_fill_time[edge_id] = current_time
        edge_water_height[edge_id] = H

        # Find downstream node
        node1, node2, length = edge_info[edge_id]
        downstream_node = node2 if node_reach_time[node2] > node_reach_time[node1] else node1

        # Find all outgoing edges from downstream node
        outgoing_edges = []
        for next_neighbor, next_edge_id, next_length in graph[downstream_node]:
            if next_edge_id != edge_id and not edge_is_filled[next_edge_id]:
                outgoing_edges.append((next_edge_id, next_neighbor, next_length))

        if outgoing_edges:
            total_flow = sum(f for _, f in edge_flow_sources[edge_id])
            flow_per_edge = total_flow / len(outgoing_edges)

            for next_edge_id, next_node, next_length in outgoing_edges:
                if not edge_has_reached_min[next_edge_id]:
                    volume_needed = W * h0 * next_length
                    next_reach_time = current_time + volume_needed / flow_per_edge
                    heapq.heappush(pq, (next_reach_time, 'reach', next_edge_id, next_node, flow_per_edge, 'filled'))
                else:
                    remaining_volume = W * (H - edge_water_height[next_edge_id]) * next_length
                    new_fill_time = current_time + remaining_volume / flow_per_edge
                    heapq.heappush(pq, (new_fill_time, 'fill', next_edge_id, next_node, flow_per_edge, 'filled'))

# 7) Prepare results
results_edges = []
for edge_id, (node1, node2, length) in edge_info.items():
    reach_min = edge_reach_time[edge_id] / 60 if edge_reach_time[edge_id] < float('inf') else np.nan
    fill_min = edge_fill_time[edge_id] / 60 if edge_fill_time[edge_id] < float('inf') else np.nan
    results_edges.append({
        'Edge': edge_id,
        'From_Node': node1,
        'To_Node': node2,
        'Length_m': round(length, 2),
        'Reach_Time_min': round(reach_min, 2) if not np.isnan(reach_min) else np.nan,
        'Fill_Time_min': round(fill_min, 2) if not np.isnan(fill_min) else np.nan
    })

results_nodes = []
for node_id in node_map:
    if node_id == burst_node_id1:
        reach_min = burst_time1 / 60
    elif node_id == burst_node_id2:
        reach_min = burst_time2 / 60
    else:
        reach_min = node_reach_time[node_id] / 60 if node_reach_time[node_id] < float('inf') else np.nan
    results_nodes.append({
        'Node': node_id,
        'Reach_Time_min': round(reach_min, 2) if not np.isnan(reach_min) else np.nan
    })

# 8) Export to Excel
df_edges = pd.DataFrame(results_edges)
df_nodes = pd.DataFrame(results_nodes)

df_edges.to_excel('result-3-1-edge.xlsx', index=False)
df_nodes.to_excel('result-3-1-node.xlsx', index=False)

print("Two-burst simulation completed!")
print(f"First burst point inserted at position {pos1:.3f}")
print(f"Second burst point inserted at position {pos2:.3f}")

# Print statistics
reachable_edges = sum(1 for t in edge_reach_time.values() if t < float('inf'))
filled_edges = sum(1 for t in edge_fill_time.values() if t < float('inf'))
total_edges = len(edge_info)

print(f"Reachable edges: {reachable_edges}/{total_edges} ({reachable_edges/total_edges*100:.1f}%)")
print(f"Filled edges: {filled_edges}/{total_edges} ({filled_edges/total_edges*100:.1f}%)")

reachable_nodes = sum(1 for t in node_reach_time.values() if t < float('inf'))
total_nodes = len(node_map)
print(f"Reachable nodes: {reachable_nodes}/{total_nodes} ({reachable_nodes/total_nodes*100:.1f}%)")
