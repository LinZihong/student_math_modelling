    waterFile = 'result1-1.xlsx';
    exitCoords = [3252.16,3326.63,10.00;

segCount = 20;

v_with = 2.0;
v_against = 1.0;

function [path_idx, best_time] = find_best_path( start_idx, exit_indices, ...
        N, M, edges, edgeFlowDir, nodeArrivalTime, neighbors, neighborEdges, neighborDir, ...
        segCount, v_dry, v_with, v_against, level_block, alert_delay, global_water_start)

    INF_time = 1e12;
    best_time = Inf;
    path_idx = [];
    dist = Inf(N,1);
    prev = zeros(N,1);
    visited = false(N,1);
    start_time = global_water_start + alert_delay;
    dist(start_idx) = start_time;

    heap_times = []; heap_nodes = [];
    [heap_times, heap_nodes] = heap_push(heap_times, heap_nodes, start_time, start_idx);
    while true
        [tcur, cur, heap_times, heap_nodes] = heap_pop(heap_times, heap_nodes);
        if isempty(tcur), break; end
        if visited(cur), continue; end
        visited(cur) = true;
        if any(exit_indices == cur)
            best_time = tcur;

            path_idx = cur;
            cur2 = cur;
            while cur2 ~= start_idx
                p = prev(cur2);
                if p == 0, break; end
                path_idx = [p, path_idx];
                cur2 = p;
            end
            return;
        end

        nbList = neighbors{cur};
        edgesList = neighborEdges{cur};
        dirList = neighborDir{cur};
        if isempty(nbList), continue; end
        for j = 1:length(nbList)
            neighbor = nbList(j);
            ek = edgesList(j);
            dirSign = dirList(j);
            if visited(neighbor), continue; end
            enter_time = tcur;
            t_need = compute_edge_travel_time(ek, enter_time, dirSign, edges, edgeFlowDir, nodeArrivalTime, segCount, v_dry, v_with, v_against, level_block);
            if ~isfinite(t_need), continue; end
            arrival = enter_time + t_need;
            if arrival < dist(neighbor)
                dist(neighbor) = arrival;
                prev(neighbor) = cur;
                [heap_times, heap_nodes] = heap_push(heap_times, heap_nodes, arrival, neighbor);
            end
        end
    end

    best_time = Inf;
    path_idx = [];
end

function t = compute_edge_travel_time(k, t_enter, dirSign, edges, edgeFlowDir, nodeArrivalTime, segCount, v_dry, v_with, v_against, level_block)
    len = edges(k,3);
    seg_len = len / segCount;

    if edgeFlowDir(k) == +1
        fillStart = nodeArrivalTime(edges(k,1));
    else
        fillStart = nodeArrivalTime(edges(k,2));
    end
    fillDuration = edges(k,4); % minutes
    t = 0.0;
    cur_t = t_enter;
    for s = 1:segCount

        if isinf(fillDuration)
            level = double(cur_t >= fillStart);
        else
            level = (cur_t - fillStart) / fillDuration;
        end
        if level < 0, level = 0; end
        if level > 1, level = 1; end

        if dirSign == +1
            workerWith = (edgeFlowDir(k) == +1);
        else
            workerWith = (edgeFlowDir(k) == -1);
        end

        if level == 0
            speed = v_dry;
        elseif level <= level_block
            if workerWith, speed = v_with; else speed = v_against; end
        else
            t = Inf; return;
        end

        dt = (seg_len / speed) / 60.0;
        cur_t = cur_t + dt;
        t = t + dt;
        if ~isinf(fillDuration)
            level_new = (cur_t - fillStart) / fillDuration;
            if level_new > level_block
                t = Inf; return;
            end
        end
    end
end

function [heap_times, heap_nodes] = heap_push(heap_times, heap_nodes, time_val, node_idx)
    heap_times(end+1) = time_val;
    heap_nodes(end+1) = node_idx;
    i = numel(heap_times);
    while i > 1
        parent = floor(i/2);
        if heap_times(parent) <= heap_times(i), break; end

        [heap_times(parent), heap_times(i)] = deal(heap_times(i), heap_times(parent));
        [heap_nodes(parent), heap_nodes(i)] = deal(heap_nodes(i), heap_nodes(parent));
        i = parent;
    end
end

function [time_val, node_idx, heap_times, heap_nodes] = heap_pop(heap_times, heap_nodes)
    if isempty(heap_times)
        time_val = []; node_idx = []; return;
    end
    time_val = heap_times(1);
    node_idx = heap_nodes(1);
    last = numel(heap_times);
    if last == 1
        heap_times = []; heap_nodes = []; return;
    end
    heap_times(1) = heap_times(last);
    heap_nodes(1) = heap_nodes(last);
    heap_times(last) = []; heap_nodes(last) = [];

    i = 1;
    while true
        left = 2*i; right = left + 1; smallest = i;
        if left <= numel(heap_times) && heap_times(left) < heap_times(smallest), smallest = left; end
        if right <= numel(heap_times) && heap_times(right) < heap_times(smallest), smallest = right; end
        if smallest == i, break; end
        [heap_times(i), heap_times(smallest)] = deal(heap_times(smallest), heap_times(i));
        [heap_nodes(i), heap_nodes(smallest)] = deal(heap_nodes(smallest), heap_nodes(i));
        i = smallest;
    end
end
